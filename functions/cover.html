type CreateTransactionRequest struct {
        Amount      float64 `json:"amount"`
        Category    string  `json:"category"`
        Description string  `json:"description"`
        Date        string  `json:"date"`
}

type TransactionResponse struct {
        ID          int     `json:"id"`
        Amount      float64 `json:"amount"`
        Category    string  `json:"category"`
        Description string  `json:"description"`
        Date        string  `json:"date"`
}

type CreateBudgetRequest struct {
        Category string  `json:"category"`
        Limit    float64 `json:"limit"`
}

type BudgetResponse struct {
        Category string  `json:"category"`
        Limit    float64 `json:"limit"`
        Period   string  `json:"period"`
}

func (r CreateTransactionRequest) ToTransaction() ledger.Transaction {
        return ledger.Transaction{
                Amount:      r.Amount,
                Category:    r.Category,
                Description: r.Description,
                Date:        r.Date,
        }
}

func TransactionResponseFrom(tx ledger.Transaction) TransactionResponse {
        return TransactionResponse{
                ID:          tx.ID,
                Amount:      tx.Amount,
                Category:    tx.Category,
                Description: tx.Description,
                Date:        tx.Date,
        }
}

func (r CreateBudgetRequest) ToBudget() ledger.Budget {
        return ledger.Budget{
                Category: r.Category,
                Limit:    r.Limit,
        }
}

func BudgetResponseFrom(b ledger.Budget) BudgetResponse {
        return BudgetResponse{
                Category: b.Category,
                Limit:    b.Limit,
                Period:   b.Period,
        }
}

package api

import (
        "encoding/json"
        "errors"
        "log"
        "net/http"

        "github.com/m0ssador/learning-go/functions/ledger"
)

func Transactions(w http.ResponseWriter, r *http.Request) {
        switch r.Method {
        case http.MethodPost:
                createTransaction(w, r)
        case http.MethodGet:
                listTransactions(w, r)
        default:
                writeError(w, http.StatusMethodNotAllowed, "method not allowed")
        }
}

func Budgets(w http.ResponseWriter, r *http.Request) {
        switch r.Method {
        case http.MethodPost:
                createBudget(w, r)
        case http.MethodGet:
                listBudgets(w, r)
        default:
                writeError(w, http.StatusMethodNotAllowed, "method not allowed")
        }
}

func createTransaction(w http.ResponseWriter, r *http.Request) {
        var req CreateTransactionRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                writeError(w, http.StatusBadRequest, "invalid json")
                return
        }

        tx := req.ToTransaction()
        if err := tx.Validate(); err != nil {
                writeError(w, http.StatusBadRequest, err.Error())
                return
        }

        created, err := ledger.AddTransaction(tx)
        if errors.Is(err, ledger.ErrBudgetExceeded) {
                writeError(w, http.StatusConflict, "budget exceeded")
                return
        }
        if err != nil {
                log.Printf("add transaction: %v", err)
                writeError(w, http.StatusInternalServerError, "internal server error")
                return
        }

        writeJSON(w, http.StatusCreated, TransactionResponseFrom(created))
}

func listTransactions(w http.ResponseWriter, _ *http.Request) {
        items := ledger.ListTransactions()
        out := make([]TransactionResponse, 0, len(items))
        for _, tx := range items {
                out = append(out, TransactionResponseFrom(tx))
        }
        writeJSON(w, http.StatusOK, out)
}

func createBudget(w http.ResponseWriter, r *http.Request) {
        var req CreateBudgetRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                writeError(w, http.StatusBadRequest, "invalid json")
                return
        }

        budget := req.ToBudget()
        if err := budget.Validate(); err != nil {
                writeError(w, http.StatusBadRequest, err.Error())
                return
        }
        if err := ledger.SetBudget(budget); err != nil {
                log.Printf("set budget: %v", err)
                writeError(w, http.StatusInternalServerError, "internal server error")
                return
        }

        writeJSON(w, http.StatusCreated, BudgetResponseFrom(budget))
}

func listBudgets(w http.ResponseWriter, _ *http.Request) {
        items := ledger.ListBudgets()
        out := make([]BudgetResponse, 0, len(items))
        for _, b := range items {
                out = append(out, BudgetResponseFrom(b))
        }
        writeJSON(w, http.StatusOK, out)
}

func LogRequests(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                next.ServeHTTP(w, r)
                log.Printf("%s %s %s", r.Method, r.URL.Path, time.Since(start))
        })
}

)

type errorBody struct {
        Error string `json:"error"`
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        if err := json.NewEncoder(w).Encode(v); err != nil {
                log.Printf("write json: %v", err)
        }
}

func writeError(w http.ResponseWriter, status int, message string) {
        writeJSON(w, status, errorBody{Error: message})
}

package ledger

import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "sort"
)

type Budget struct {
        Category string
        Limit    float64
        Period   string
}

func (b Budget) Validate() error {
        if b.Limit <= 0 {
                return errors.New("budget limit must be positive")
        }
        if b.Category == "" {
                return errors.New("budget category must not be empty")
        }
        return nil
}

func SetBudget(b Budget) error {
        if err := b.Validate(); err != nil {
                return err
        }

        mu.Lock()
        defer mu.Unlock()

        if budgets == nil {
                budgets = make(map[string]Budget)
        }
        budgets[b.Category] = b
        return nil
}

func ListBudgets() []Budget {
        mu.Lock()
        defer mu.Unlock()

        result := make([]Budget, 0, len(budgets))
        for _, b := range budgets {
                result = append(result, b)
        }
        sort.Slice(result, func(i, j int) bool {
                return result[i].Category < result[j].Category
        })
        return result
}

func LoadBudgets(r io.Reader) error {
        data, err := io.ReadAll(r)
        if err != nil {
                return fmt.Errorf("read budgets: %w", err)
        }

        var items []Budget
        if err := json.Unmarshal(data, &items); err != nil {
                return fmt.Errorf("parse budgets: %w", err)
        }

        for i, b := range items {
                if err := b.Validate(); err != nil {
                        return fmt.Errorf("budget %d: %w", i+1, err)
                }
        }
        for _, b := range items {
                if err := SetBudget(b); err != nil {
                        return err
                }
        }
        return nil
}

import "sync"

var (
        mu           sync.Mutex
        transactions []Transaction
        budgets      map[string]Budget
)

func Reset() {
        mu.Lock()
        defer mu.Unlock()
        transactions = nil
        budgets = nil

package ledger

import "errors"

var ErrBudgetExceeded = errors.New("budget exceeded")

type Transaction struct {
        ID          int
        Amount      float64
        Category    string
        Description string
        Date        string
}

func (t Transaction) Validate() error {
        if t.Amount == 0 {
                return errors.New("transaction amount must not be zero")
        }
        if t.Amount < 0 {
                return errors.New("transaction amount must not be negative")
        }
        if t.Category == "" {
                return errors.New("transaction category must not be empty")
        }
        return nil
}

func AddTransaction(tx Transaction) (Transaction, error) {
        if err := tx.Validate(); err != nil {
                return Transaction{}, err
        }

        mu.Lock()
        defer mu.Unlock()

        if b, ok := budgets[tx.Category]; ok {
                var spent float64
                for _, existing := range transactions {
                        if existing.Category == tx.Category {
                                spent += existing.Amount
                        }
                }
                if spent+tx.Amount > b.Limit {
                        return Transaction{}, ErrBudgetExceeded
                }
        }

        tx.ID = len(transactions) + 1
        transactions = append(transactions, tx)
        return tx, nil
}

func ListTransactions() []Transaction {
        mu.Lock()
        defer mu.Unlock()

        result := make([]Transaction, len(transactions))
        copy(result, transactions)
        return result
}
